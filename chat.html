import { initializeApp } from 'firebase/app'; import { getAuth, signInAnonymously } from 'firebase/auth'; import { getFirestore, collection, addDoc, serverTimestamp, query, orderBy, onSnapshot } from 'firebase/firestore'; // Firebase konfiguratsiyasi const firebaseConfig = { apiKey: "YOUR_API_KEY", // Firebase konsolidan loyiha sozlamalaridan olasiz authDomain: "smartalim-chat.firebaseapp.com", projectId: "smartalim-chat", storageBucket: "smartalim-chat.appspot.com", messagingSenderId: "975545287065", appId: "YOUR_APP_ID" // Firebase konsolidan olasiz }; const app = initializeApp(firebaseConfig); const auth = getAuth(app); const db = getFirestore(app); // Anonim tarzda tizimga kirish signInAnonymously(auth) .then(() => { console.log("Tizimga kirildi!"); loadMessages(); }) .catch((error) => { console.error("Xatolik yuz berdi:", error); }); // Xabar yuborish funksiyasi async function sendMessage(text) { if (!text.trim()) return; try { await addDoc(collection(db, "chats"), { text: text, createdAt: serverTimestamp(), userId: auth.currentUser.uid }); } catch (error) { console.error("Xabarni yuborishda xatolik:", error); } } // Xabarlarni real vaqt rejimida yuklash function loadMessages() { const q = query(collection(db, "chats"), orderBy("createdAt", "asc")); onSnapshot(q, (snapshot) => { const messagesList = document.getElementById("messages"); messagesList.innerHTML = ""; snapshot.forEach((doc) => { const msg = doc.data(); const li = document.createElement("li"); li.textContent = `${msg.userId.substring(0, 5)}: ${msg.text}`; messagesList.appendChild(li); }); }); }
